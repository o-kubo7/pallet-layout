# 配置編集の「戻す・進む」 実装計画

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 配置編集タブで、直前の操作（荷物の移動・マスの選択）を1つずつ取り消し、やり直せるようにする。

**Architecture:** 履歴は時間帯ごとの配列とカーソルで持つ。ステップは2種類で、選択だけが変わったステップは配置（`sp`）を前のステップと同じ参照にしてコピーしない。履歴のロジックはDOMに触らない純粋関数として切り出し、既存テストの `functionSource()` で取り出して単体で検証する。

**Tech Stack:** 素の JavaScript（`files/index.html` に埋め込み、ビルドなし）、`node:test` によるソース検査＋関数取り出し実行

**Spec:** `docs/superpowers/specs/2026-09-21-undo-redo-design.md`

// 関数をソースから切り出す。既存テスト（tests/sheet-placement.test.js:339）と同じ実装。
function functionSource(name) {
  const start = source.indexOf(`function ${name}(`);
  assert.notEqual(start, -1, `${name} must exist`);
  const brace = source.indexOf("{", start);
  let depth = 0, quote = null, escaped = false;
  for (let i = brace; i < source.length; i++) {
    const char = source[i];
    if (quote) {
      if (escaped) escaped = false;
      else if (char === "\\") escaped = true;
      else if (char === quote) quote = null;
      continue;
    }
    if (char === "'" || char === '"' || char === "`") { quote = char; continue; }
    if (char === "{") depth++;
    if (char === "}" && --depth === 0) return source.slice(start, i + 1);
  }
  throw new Error(`${name} has no closing brace`);
}

`files/index.html` の `function clone(o){...}`（`files/index.html:6136`）の**直後**に置く。
`clone()` より後に置くのは、この節がユーティリティ関数の並びだから。
履歴の関数は `clone()` を使わない（呼ぶ側が使う）。

```js
/* ---------- 配置編集の履歴（戻す・進む） ----------
   steps[cursor] が「いまの状態」。cursor が -1 なら履歴なし。
   ステップは2種類ある。
     kind:"move"   … 荷物を動かした。sp は新しい配列
     kind:"select" … マスを選んだ・外した。sp は前のステップと同じ参照
   選択だけが変わったときに sp をコピーしないので、選択を何十回繰り返しても
   メモリは増えない。設計書: docs/superpowers/specs/2026-09-21-undo-redo-design.md §4
   DOM もグローバル変数も見ない。呼ぶ側（pushMoveStep など）が面倒を見る。 */
const HISTORY_LIMIT = 50;
function historyMake(){ return {steps:[], cursor:-1}; }
function historyPush(hist, step){
  // やり直せる分（cursor より後ろ）を捨ててから足す。
  // 戻ってから別の操作をしたら、元の枝には戻れない。
  hist.steps.length = hist.cursor + 1;
  hist.steps.push(step);
  // 上限を超えたぶんを先頭から落とし、cursor も同じ数だけ詰める
  const over = hist.steps.length - HISTORY_LIMIT;
  if(over > 0){ hist.steps.splice(0, over); }
  hist.cursor = hist.steps.length - 1;
  return hist;
}
function historyCanUndo(hist){ return !!hist && hist.cursor > 0; }
function historyCanRedo(hist){ return !!hist && hist.cursor >= 0 && hist.cursor < hist.steps.length - 1; }
function historyUndo(hist){
  if(!historyCanUndo(hist)) return null;
  hist.cursor--;
  return hist.steps[hist.cursor];
}
function historyRedo(hist){
  if(!historyCanRedo(hist)) return null;
  hist.cursor++;
  return hist.steps[hist.cursor];
}
```

```js
/* いまの時間帯の履歴。保存はしない（読み込み直すと消える）。
   あさ・ひるを別々に持つので、切り替えて戻ってきても、その時間帯でした操作を戻せる。 */
let histories = { am:historyMake(), pm:historyMake() };
function activeHistory(){ return histories[activeTiming]; }
function clearHistory(){ histories[activeTiming] = historyMake(); }
function selSnapshot(){ return {lotId: sel.lotId, cells:[...sel.cells]}; }
/* いまのカーソルが指すステップの配置。選択だけのステップはこれを使い回す。 */
function currentStepSp(){
  const h = activeHistory();
  return h.cursor >= 0 ? h.steps[h.cursor].sp : null;
}
/* 荷物を動かしたときに積む。spBefore は動かす前の配置（呼ぶ側が控える）。
   履歴が空のときは、先に「動かす前」を1件積む。そうしないと 1 件目で
   cursor が 0 になり、historyCanUndo が false のままで戻せない。 */
function pushMoveStep(spBefore, selBefore){
  const h = activeHistory();
  // 履歴が空なら「動かす前」を土台として先に積む。選択も動かす前のものを持たせるので、
  // 戻ったときにそのまま選び直さずに操作を続けられる
  if(h.cursor < 0 && spBefore){
    historyPush(h, {kind:"move", sel:selBefore, sp:spBefore});
  }
  // 移動後は clearSel() で選択が消える。ここで移動前の選択を持たせると、
  // やり直したときに「移動元のマス」のキーが戻り、荷物が無いのに
  // 「✋ Nマス選択中」と出る。実体のない選択を作らないため空にする
  historyPush(h, {kind:"move", sel:{lotId:null, cells:[]}, sp:snapshotSpaces(lastSp)});
}
/* 選択が変わったときに積む。配置は変わらないので、いまのステップの sp を
   そのまま指す（コピーしない）。履歴が空のときは、いまの配置を基準として持つ。 */
function pushSelectStep(){
  const h = activeHistory();
  const sp = currentStepSp() || (lastSp ? snapshotSpaces(lastSp) : null);
  if(!sp) return;                       // まだ配置が無いときは積まない
  // 履歴が空なら「選ぶ前」を土台として先に積む。そうしないと cursor が 0 のままで
  // historyCanUndo が false になり、最初の1回だけ戻せない（移動と挙動が変わる）
  if(h.cursor < 0){
    historyPush(h, {kind:"select", sel:{lotId:null, cells:[]}, sp:sp});
  }
  historyPush(h, {kind:"select", sel:selSnapshot(), sp:sp});
}
```

`let histories = ...` は `activeTiming` より後に置く必要がある。
`activeTiming` は `files/index.html:1244` で宣言済みなので、Task 1 の位置（`clone()` の後、6136付近）なら順序の問題はない。

/* 履歴のステップを画面に反映する。kind で分けない。
   選択だけのステップは sp が前と同じ参照なので、結果として配置は変わらない。
   分岐を作らないほうが間違えにくい。設計書 §6 */
function applyHistoryStep(step){
  if(!step) return;

  lastSp = clone(step.sp);

   aria-label と title は変えない。記号だけにしても読み上げと長押しで意味が分かる。 */
function toggleUndoIcon(){
  const on=document.getElementById("undoIconChk").checked;
  saveData(STORE_KEY.undoIcon, on);
  applyUndoIcon(on);

function applyUndoIcon(on){
  const u=document.getElementById("undoBtn"), r=document.getElementById("redoBtn");
  if(u) u.textContent = on ? "↩" : "↩ 戻す";
  if(r) r.textContent = on ? "↪" : "進む ↪";
}
